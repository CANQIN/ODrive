cmake_minimum_required(VERSION 3.20)
project(fibre_client_example LANGUAGES C CXX)

set(FIBRE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../.." CACHE PATH "Fibre 包源码目录")
set(FIBRE_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(FIBRE_BUILD_CLIENT ON CACHE BOOL "" FORCE)
set(FIBRE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
if(MSVC)
    add_compile_options(/Zc:__cplusplus /utf-8)
endif()
add_subdirectory("${FIBRE_SOURCE_DIR}" fibre)

add_executable(fibre_client_example main.cpp)
set_target_properties(fibre_client_example PROPERTIES OUTPUT_NAME fibre_client)
target_link_libraries(fibre_client_example PRIVATE fibre::client)
target_include_directories(fibre_client_example PRIVATE transports)
if(WIN32)
    target_link_libraries(fibre_client_example PRIVATE ws2_32)
endif()
